Q1 · 手写 flatten，支持指定深度

📌 场景

后端返回嵌套分类树要转成一维 id 列表——flatten([1, [2, [3]], 4], 2)

查看答案 
function flatten(arr, depth = Infinity) {
  const result = [];
  for (const item of arr) {
    if (Array.isArray(item) && depth > 0) {
      result.push(...flatten(item, depth - 1));
    } else {
      result.push(item);
    }
  }
  return result;
}

// 简易版：arr.flat(Infinity)

Q3 · debounce 和 throttle 手写

📌 场景

搜索框联想 vs 滚动加载——各用哪个？完整实现见 闭包 Q5/Q6。

查看答案 
function debounce(fn, delay) {
  let timer = null;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

function throttle(fn, interval) {
  let last = 0;
  return function (...args) {
    const now = Date.now();
    if (now - last >= interval) {
      last = now;
      fn.apply(this, args);
    }
  };
}

Q4 · 深拷贝（含循环引用）

完整版见 类型&拷贝 Q8。

查看答案 
function deepClone(val, cache = new WeakMap()) {
  if (val === null || typeof val !== 'object') return val;
  if (cache.has(val)) return cache.get(val);
  const result = Array.isArray(val) ? [] : {};
  cache.set(val, result);
  for (const k of Reflect.ownKeys(val)) {
    result[k] = deepClone(val[k], cache);
  }
  return result;
}

Function.prototype.myBind = function (ctx, ...preset) {
  const fn = this;
  return function bound(...args) {
    const isNew = this instanceof bound;
    return fn.apply(isNew ? this : ctx, preset.concat(args));
  };
};

Q6 · 手写 curry

查看答案 
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return (...next) => curried.apply(this, args.concat(next));
  };
}

Q8 · 发布订阅 EventEmitter

📌 场景

上传任务中心：进度、完成、失败要通知多个 UI 组件——实现简易事件总线。

查看答案 
class EventEmitter {
  #map = new Map();

  on(event, fn) {
    if (!this.#map.has(event)) this.#map.set(event, new Set());
    this.#map.get(event).add(fn);
    return () => this.off(event, fn);
  }

  off(event, fn) {
    this.#map.get(event)?.delete(fn);
  }

  emit(event, ...args) {
    this.#map.get(event)?.forEach((fn) => fn(...args));
  }

  once(event, fn) {
    const wrap = (...args) => {
      this.off(event, wrap);
      fn(...args);
    };
    this.on(event, wrap);
  }
}

Q11 · 手写 v-permission 权限指令

📌 场景

没 bpo:staff:delete 权限时移除删除按钮 DOM——

查看答案 
function installPermission(app, getPerms) {
  app.directive('permission', {
    mounted(el, binding) {
      const need = binding.value
      const perms = getPerms()
      const ok = Array.isArray(need)
        ? need.some(p => perms.includes(p))
        : perms.includes(need)
      if (!ok) el.remove()
    }
  })
}

Q12 · 手写配置驱动表格列渲染（简化 ProTable）

📌 场景

columns 配置 + slot 自定义列——核心 render 逻辑怎么写？

查看答案 
function renderColumns(columns, slots) {
  return columns.map(col => {
    if (col.slot && slots[col.slot]) {
      return { ...col, render: (row) => slots[col.slot]({ row }) }
    }
    return { ...col, render: (row) => row[col.prop] }
  })
}

Vue 里用 <template v-for> + 动态 slot 名 #[col.slot] 透传。
